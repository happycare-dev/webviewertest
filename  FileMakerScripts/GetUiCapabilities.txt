# GetUiCapabilities — tells Web Viewer whether to show Edit/Delete (Perform JS → receiveUiCapabilities, object web)
#
# LoginValidate must set $$WebViewerEmployeeLoggedIn after a successful Re-Login.

If [ not $$WebViewerEmployeeLoggedIn = 1 ]
	Set Variable [ $json ; Value: JSONSetElement ( "{}" ; [ "canEditDelete" ; False ; JSONBoolean ] ; [ "error" ; "login_required" ; JSONString ] ) ]
	Set Variable [ $jsonText ; Value: GetAsText ( $json ) ]
	Perform JavaScript in Web Viewer [ Object Name: "web" ; Function Name: "receiveUiCapabilities" ; Parameters: $jsonText ]
	Exit Script [ Text Result: False ]
End If

Set Variable [ $recordAccess ; Value: Get ( RecordAccess ) ]
Set Variable [ $layoutAccess ; Value: Get ( LayoutAccess ) ]
Set Variable [ $allow ; Value: Case ( GetAsNumber ( $recordAccess ) = 2 and GetAsNumber ( $layoutAccess ) = 2 ; True ; False ) ]
Set Variable [ $json ; Value: JSONSetElement ( "{}" ; [ "canEditDelete" ; $allow ; JSONBoolean ] ; [ "recordAccess" ; $recordAccess ; JSONNumber ] ; [ "layoutAccess" ; $layoutAccess ; JSONNumber ] ) ]
Set Variable [ $jsonText ; Value: GetAsText ( $json ) ]
Perform JavaScript in Web Viewer [ Object Name: "web" ; Function Name: "receiveUiCapabilities" ; Parameters: $jsonText ]
Exit Script [ Text Result: True ]
