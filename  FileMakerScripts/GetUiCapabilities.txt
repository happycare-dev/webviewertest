# GetUiCapabilities — tells Web Viewer whether to show Edit/Delete (Perform JS → receiveUiCapabilities, object web)
#
# LoginValidate sets $$wvLoginAccount / $$wvLoginPrivilegeSet / $$wvLoginCanEditDelete after password validation.
# Do not derive UI permissions from Get(AccountPrivilegeSetName): that is the FileMaker session account, not the logged-in employee.

If [ IsEmpty ( $$wvLoginAccount ) ]
	Set Variable [ $json ; Value: JSONSetElement ( "{}" ; [ "canEditDelete" ; False ; JSONBoolean ] ; [ "authenticated" ; False ; JSONBoolean ] ; [ "message" ; "login required" ; JSONString ] ) ]
	Perform JavaScript in Web Viewer [ Object Name: "web" ; Function Name: "receiveUiCapabilities" ; Parameters: GetAsText ( $json ) ]
	Exit Script [ Text Result: False ]
End If

Set Variable [ $json ; Value: JSONSetElement ( "{}" ; [ "canEditDelete" ; $$wvLoginCanEditDelete ; JSONBoolean ] ; [ "authenticated" ; True ; JSONBoolean ] ; [ "accountName" ; $$wvLoginAccount ; JSONString ] ; [ "privilegeSetName" ; $$wvLoginPrivilegeSet ; JSONString ] ) ]
Set Variable [ $jsonText ; Value: GetAsText ( $json ) ]
Perform JavaScript in Web Viewer [ Object Name: "web" ; Function Name: "receiveUiCapabilities" ; Parameters: $jsonText ]
Exit Script [ Text Result: True ]
