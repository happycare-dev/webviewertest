# LoginValidate in file WebViewerTest
#
# Web Viewer（オブジェクト名 web）からのパラメータ JSON:
#   { "account": "...", "password": "..." }
# 検証: EmployeeM の「アカウント」一致、在籍フラグ=1、パスワード一致 → のみレイアウト「WebViewerTest」へ移動。
# 失敗時は receiveLoginResult に JSON（GetAsText 推奨）。レイアウト移動は $loginSuccess = 1 のときのみ 1 か所で実行。
#
# 前提:
# - Data API が EmployeeM を参照できること
# - 遷移先レイアウト名が「WebViewerTest」であること（異なる場合は最後の Go to Layout のみ修正）
# - Login レイアウトの Web Viewer オブジェクト名が「web」であること

Set Variable [ $param ; Value: Get ( ScriptParameter ) ]
Set Variable [ $account ; Value: JSONGetElement ( $param ; "account" ) ]
Set Variable [ $password ; Value: JSONGetElement ( $param ; "password" ) ]

# A new login attempt must not inherit capabilities from a previous successful login in the same FileMaker session.
Set Variable [ $$wvLoginAccount ; Value: "" ]
Set Variable [ $$wvLoginPrivilegeSet ; Value: "" ]
Set Variable [ $$wvLoginCanEditDelete ; Value: False ]

# 0 = 失敗（レイアウト移動しない）、1 = すべて合格
Set Variable [ $loginSuccess ; Value: 0 ]
Set Variable [ $errorCode ; Value: "" ]
Set Variable [ $apiMsg ; Value: "" ]

# 従業員マスタをアカウントで検索（先頭1件）
Set Variable [ $request ; Value: JSONSetElement ( "" ; [ "layouts" ; "EmployeeM" ; JSONString ] ; [ "limit" ; 1 ; JSONNumber ] ; [ "offset" ; 1 ; JSONNumber ] ; [ "query[0].アカウント" ; $account ; JSONString ] ) ]
Execute FileMaker Data API [ Select ; Target: $result ; $request ]

# --- Data API 応答 ---
# 検索で 0 件のとき、環境によっては code=0 + foundCount=0 ではなく、
# code=401 や message「No records match…」だけが返ることがある。その場合も no_account と同じ扱いにする。
Set Variable [ $apiCode ; Value: JSONGetElement ( $result ; "messages[0].code" ) ]
If [ not IsEmpty ( $apiCode ) and $apiCode ≠ "0" ]
	Set Variable [ $apiMsg ; Value: JSONGetElement ( $result ; "messages[0].message" ) ]
	If [ GetAsNumber ( $apiCode ) = 401 or PatternCount ( Lower ( $apiMsg ) ; "no records match" ) > 0 ]
		Set Variable [ $errorCode ; Value: "no_account" ]
	Else
		Set Variable [ $errorCode ; Value: "api_error" ]
	End If
Else
	# --- 件数（アカウント存在）---
	Set Variable [ $foundCount ; Value: JSONGetElement ( $result ; "response.dataInfo.foundCount" ) ]
	If [ IsEmpty ( $foundCount ) or GetAsNumber ( $foundCount ) < 1 ]
		Set Variable [ $errorCode ; Value: "no_account" ]
	Else
		Set Variable [ $flag ; Value: JSONGetElement ( $result ; "response.data[0].fieldData.在籍フラグ" ) ]
		Set Variable [ $storedPw ; Value: JSONGetElement ( $result ; "response.data[0].fieldData.パスワード" ) ]
		# --- 在籍のみ ---
		If [ GetAsNumber ( $flag ) ≠ 1 ]
			Set Variable [ $errorCode ; Value: "not_active" ]
		Else
			# --- パスワード（大文字小文字・内容を Exact で一致）---
			If [ not Exact ( $storedPw ; $password ) ]
				Set Variable [ $errorCode ; Value: "wrong_password" ]
			Else
				Set Variable [ $loginSuccess ; Value: 1 ]
			End If
		End If
	End If
End If

# --- 失敗: コールバックのみ。レイアウトは変更しない ---
If [ $loginSuccess ≠ 1 ]
	If [ $errorCode = "api_error" ]
		Set Variable [ $out ; Value: JSONSetElement ( "{}" ; [ "ok" ; False ; JSONBoolean ] ; [ "error" ; "api_error" ; JSONString ] ; [ "message" ; $apiMsg ; JSONString ] ) ]
	Else If [ $errorCode = "no_account" ]
		Set Variable [ $out ; Value: JSONSetElement ( "{}" ; [ "ok" ; False ; JSONBoolean ] ; [ "error" ; "no_account" ; JSONString ] ) ]
	Else If [ $errorCode = "not_active" ]
		Set Variable [ $out ; Value: JSONSetElement ( "{}" ; [ "ok" ; False ; JSONBoolean ] ; [ "error" ; "not_active" ; JSONString ] ) ]
	Else If [ $errorCode = "wrong_password" ]
		Set Variable [ $out ; Value: JSONSetElement ( "{}" ; [ "ok" ; False ; JSONBoolean ] ; [ "error" ; "wrong_password" ; JSONString ] ) ]
	Else
		Set Variable [ $out ; Value: JSONSetElement ( "{}" ; [ "ok" ; False ; JSONBoolean ] ; [ "error" ; "no_account" ; JSONString ] ) ]
	End If
	Perform JavaScript in Web Viewer [ Object Name: "web" ; Function Name: "receiveLoginResult" ; Parameters: GetAsText ( $out ) ]
	Exit Script [ Text Result: False ]
End If

# --- 成功時のみ（上で Exit 済みのため、ここに到達するのは $loginSuccess = 1 のときだけ）---
# Web Viewer scripts must trust this validated login state, not the ambient FileMaker account directly.
Set Variable [ $$wvLoginAccount ; Value: $account ]
Set Variable [ $$wvLoginPrivilegeSet ; Value: Get ( AccountPrivilegeSetName ) ]
Set Variable [ $$wvLoginCanEditDelete ; Value: Case (
	$$wvLoginPrivilegeSet = "[Full Access]" ; True ;
	$$wvLoginPrivilegeSet = "Admin" ; True ;
	$$wvLoginPrivilegeSet = "Manager" ; True ;
	False
) ]
# Web Viewer 側で「ログイン中」のまま固まらないよう、先に成功コールバックで busy を解除してからレイアウト移動
Set Variable [ $okOut ; Value: JSONSetElement ( "{}" ; [ "ok" ; True ; JSONBoolean ] ) ]
Perform JavaScript in Web Viewer [ Object Name: "web" ; Function Name: "receiveLoginResult" ; Parameters: GetAsText ( $okOut ) ]
Go to Layout [ "WebViewerTest" ; Animation: None ]
Exit Script [ Text Result: True ]
